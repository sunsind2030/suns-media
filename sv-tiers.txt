  function svCartTiers(){
    var form=document.querySelector('[data-store="cart-form"]'); var anchor=document.querySelector('.cart-totals-container');
    if(!form||!anchor||document.querySelector('.sv-tier')) return;
    var T=[[75000,5],[99500,10],[149500,15],[199500,20]], FREE=99500;
    var st=document.createElement('style');
    st.textContent='.sv-tier{border:1px solid #e6e6e6;padding:16px 16px 12px;margin:0 0 20px;background:#fff}.sv-tier-k{font:500 11px/1 "Oswald","Arial Narrow",Arial,sans-serif;letter-spacing:.14em;text-transform:uppercase;color:#000;margin-bottom:8px}.sv-tier-m{font:400 13px/1.45 "Inter","Helvetica Neue",Arial,sans-serif;color:#444;margin-bottom:14px}.sv-tier-m b{color:#000;font-weight:700}.sv-tier-bar{position:relative;height:4px;background:#ececec;margin-bottom:8px;overflow:hidden}.sv-tier-fill{position:absolute;left:0;top:0;bottom:0;width:0;background:#000;transition:width .5s ease}.sv-tier-steps{display:grid;grid-template-columns:repeat(4,1fr)}.sv-tier-s{position:relative;text-align:right;font:400 10px/1.3 "Inter","Helvetica Neue",Arial,sans-serif;color:#aaa}.sv-tier-s b{display:block;font:500 12px/1.2 "Oswald","Arial Narrow",Arial,sans-serif;letter-spacing:.06em;color:#aaa}.sv-tier-s::before{content:"";position:absolute;right:0;top:-12px;width:2px;height:4px;background:#fff}.sv-tier-s:last-child::before{display:none}.sv-tier-s.on,.sv-tier-s.on b{color:#000}';
    document.head.appendChild(st);
    var box=document.createElement('div'); box.className='sv-tier';
    var steps=''; for(var i=0;i<T.length;i++){ steps+='<div class="sv-tier-s"><b>'+T[i][1]+'% OFF</b>$'+T[i][0].toLocaleString('es-AR')+'</div>'; }
    box.innerHTML='<div class="sv-tier-k">Cuanto más llevás, más ahorrás</div><div class="sv-tier-m"></div><div class="sv-tier-bar"><div class="sv-tier-fill"></div></div><div class="sv-tier-steps">'+steps+'</div>';
    anchor.parentNode.insertBefore(box,anchor);
    var msgEl=box.querySelector('.sv-tier-m'), fill=box.querySelector('.sv-tier-fill'), sEls=box.querySelectorAll('.sv-tier-s');
    function money(n){ return '$'+Math.ceil(n).toLocaleString('es-AR'); }
    function sub(){
      var e=document.querySelector('[data-store="cart-subtotal"]');
      if(e){ var m=e.textContent.replace(/\s/g,'').match(/\$([\d\.]+)(,\d+)?/); if(m) return parseFloat(m[1].replace(/\./g,'')+(m[2]?'.'+m[2].slice(1):'')); }
      if(window.LS&&LS.cart&&LS.cart.subtotal!=null) return LS.cart.subtotal/100;
      return 0;
    }
    var lastS=-1;
    function render(){
      var s=sub(); if(s===lastS) return; lastS=s;
      var cur=0, next=null, idx=-1;
      for(var i=0;i<T.length;i++){ if(s>=T[i][0]){ cur=T[i][1]; idx=i; } else if(!next){ next=T[i]; } }
      var extra=(next&&next[0]===FREE&&s<FREE)?' + envío gratis':'';
      var msg;
      if(!next) msg='<b>¡Tenés el descuento máximo: 20% OFF!</b> Se aplica solo al pagar.';
      else if(cur) msg='<b>Ya tenés '+cur+'% OFF.</b> Sumá '+money(next[0]-s)+' más y llevás <b>'+next[1]+'% OFF'+extra+'</b>.';
      else msg='Sumá <b>'+money(next[0]-s)+'</b> más y llevás <b>'+next[1]+'% OFF'+extra+'</b>.';
      msgEl.innerHTML=msg;
      var p=0, seg=100/T.length;
      for(var j=0;j<T.length;j++){
        var lo=j===0?0:T[j-1][0], hi=T[j][0];
        if(s>=hi){ p+=seg; } else { p+=seg*Math.max(0,(s-lo)/(hi-lo)); break; }
      }
      fill.style.width=Math.min(100,p)+'%';
      for(var k=0;k<sEls.length;k++){ sEls[k].classList.toggle('on',k<=idx); }
    }
    render(); setInterval(render,800);
  }
